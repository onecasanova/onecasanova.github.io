<script>
  new QRCode(document.getElementById("qrcode"), {
    text: "LESCXN-EQD3KQRK",
    width: 130,
    height: 130,
    colorDark: "#000000",
    colorLight: "#ffffff",
    correctLevel: QRCode.CorrectLevel.M
  });

  function formatIssued(d) {
    const days = ["Sun","Mon","Tue","Wed","Thu","Fri","Sat"];
    const months = ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"];
    const dayName = days[d.getDay()];
    const monthName = months[d.getMonth()];
    const date = String(d.getDate()).padStart(2, "0");
    const year = d.getFullYear();
    const hours = String(d.getHours()).padStart(2, "0");
    const mins = String(d.getMinutes()).padStart(2, "0");
    return `${dayName} ${monthName} ${date}, ${year} ${hours}:${mins}`;
  }

  function formatExpiration(d) {
    const month = String(d.getMonth() + 1).padStart(2, "0");
    const date = String(d.getDate()).padStart(2, "0");
    const year = d.getFullYear();
    const hours = String(d.getHours()).padStart(2, "0");
    const mins = String(d.getMinutes()).padStart(2, "0");
    return `${month}/${date}/${year} ${hours}:${mins}`;
  }

  function updateTicketDates() {
    const now = new Date();
    const nextDay = new Date(now);
    nextDay.setDate(nextDay.getDate() + 1);

    document.getElementById("issuedText").textContent = "Issued: " + formatIssued(now);
    document.getElementById("expirationText").textContent = "Expiration Date: " + formatExpiration(nextDay);
  }

  // Run once automatically when the page loads
  updateTicketDates();

  // Also run on tap/click
  document.getElementById("qrBlock").addEventListener("click", updateTicketDates);

</script>
